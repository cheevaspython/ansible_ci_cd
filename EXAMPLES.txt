# ============================================================================
# QUICK START COMMANDS
# ============================================================================

# INSTALLATION
pip install ansible
cd ansible
ansible-galaxy collection install -r requirements.yml

# INVENTORY SETUP
vim inventories/test/hosts
# Добавь: test-swarm-01 ansible_host=YOUR_IP ansible_user=admin_remote ansible_port=3322

# SSH KEYS
ssh-keygen -t ed25519 -f ~/.ssh/swarm_deploy
ssh-copy-id -i ~/.ssh/swarm_deploy.pub -p 3322 admin_remote@YOUR_IP

# VERIFICATION
cd scripts
chmod +x setup_environment.sh
./setup_environment.sh test

# DOCKER SECRETS (создаются ВРУЧНУЮ на Swarm manager один раз)
# См. раздел "СОЗДАНИЕ DOCKER SECRETS" ниже

# DOCKER NETWORKS (на Swarm manager)
docker network create --driver overlay internal_network
docker network create --driver overlay public_network
docker network create --driver overlay main_internal_network

# ============================================================================
# DEPLOYMENT COMMANDS
# ============================================================================

# Syntax check
ansible-playbook deploy.yml -i inventories/test --syntax-check

# Dry run
ansible-playbook deploy.yml -i inventories/test --check

# Deploy to test
ansible-playbook deploy.yml -i inventories/test

# Deploy to production
ansible-playbook deploy.yml -i inventories/production

# Run migrations
ansible-playbook migrate.yml -i inventories/test

# Rollback
ansible-playbook rollback.yml -i inventories/production

# Check status
ansible-playbook status.yml -i inventories/production

# ============================================================================
# DOCKER COMMANDS
# ============================================================================

# Service management
docker service ls
docker service ps main_backend_product
docker service logs main_backend_product --tail 100 --follow

# Stack management
docker stack ls
docker stack ps main
docker stack services main

# Get deployed image
docker service inspect main_backend_product --format '{{.Spec.TaskTemplate.ContainerSpec.Image}}'

# ============================================================================
# СОЗДАНИЕ DOCKER SECRETS (ВРУЧНУЮ, ОДИН РАЗ)
# ============================================================================

# ВАЖНО: Secrets создаются ВРУЧНУЮ на Swarm manager!
# НЕ храните пароли в скриптах или в Git!

# Создание secrets интерактивно (безопасно)
echo -n "your_postgres_user" | docker secret create postgres_user -
echo -n "your_secure_password" | docker secret create postgres_password_product -
echo -n "your_redis_password" | docker secret create redis_product_password -
echo -n "your_s3_access_key" | docker secret create s3_access_key -
echo -n "your_s3_secret_key" | docker secret create s3_secret_key -

# Проверка созданных secrets
docker secret ls

# ВАЖНО: После создания secrets НЕ МОГУТ быть прочитаны!
# Убедитесь что вы сохранили пароли в безопасном месте (KeePass, 1Password, etc.)

# ============================================================================
# ADAPTING FOR OTHER MICROSERVICES
# ============================================================================

# Example: calorie_auth

# 1. Update group_vars
sed -i 's/service_name: "product"/service_name: "auth"/' inventories/*/group_vars/all.yml
sed -i 's/service_suffix: "_product"/service_suffix: "_auth"/' inventories/*/group_vars/all.yml

# 2. Update .gitlab-ci.yml
sed -i 's/SERVICE_NAME: "product"/SERVICE_NAME: "auth"/' .gitlab-ci.yml
sed -i 's/SERVICE_SUFFIX: "_product"/SERVICE_SUFFIX: "_auth"/' .gitlab-ci.yml

# 3. Create secrets
echo 'password' | docker secret create postgres_password_auth -
echo 'password' | docker secret create redis_auth_password -

# 4. Deploy
ansible-playbook deploy.yml -i inventories/test

# ============================================================================
# TROUBLESHOOTING
# ============================================================================

# Test connectivity
ansible all -i inventories/test -m ping

# Verbose mode
ansible-playbook deploy.yml -i inventories/test -vvv

# Check Docker daemon
ansible swarm_managers -i inventories/test -m shell -a "systemctl status docker"

# Check Swarm status
ansible swarm_managers -i inventories/test -m shell -a "docker info --format '{{.Swarm.LocalNodeState}}'"
